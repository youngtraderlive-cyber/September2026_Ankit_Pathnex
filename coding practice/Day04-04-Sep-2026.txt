Jenkinsfile - Basix CI Pipeline
Pipeline {
  agent any
  stages { 
    stage('Build') {
      step {
        echo 'Building project'
       }
    }
    stage('test') {
       steps { 
         echo 'Running tests'
      }
   }
   stage('Deploy') {
     step {
         echo 'Deployment application'
               }
          }
     }
}

Ansible - Install  and Configure Nginx
- name: Install and confirure Ngix
  host: all
  become: yes
  task:
   - name: Install nginx
     yum:
      name: nginx
      state: present
   - name: Configure nginx
     template
       src: nginx.conf.j2
       dest: /ect/nginx/nginx.conf
   - name: Start nginx service
     service:
       name: nginx
       state: started
        enabled: yes
