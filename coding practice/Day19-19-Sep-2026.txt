Ansible advance monitoring logging ans security
_ name: Install prometheus for monitoring
  hosts: all
  become: yes
    taks:
      - name:  Install prometheus dependencies
        yum:
         name:
            - prometheus
            - prometheus-node-exporter
          state: present
       - name: Start prometheus service
         service:
           name: prometheus
           state: started
          enabled: yes

Kubernetes deploy fluentd for logging
apiVersion: app/V1
kind: Deployment
metadata: 
    name: fluentd
spec:
  replicas: 1
   selector: 
      matchLabels:
       app:fluentd
 template:
   metadata:
     labels:
      app: fluentd
 spec:
   container:
     - name: fluentd
       image: fluent/fluentd
       ports:
          - containerPort: 24224
---

apiVersion: v1
kind: service
metadata:
   name: fluentd
   spec:
    selector:
      app: fluentd
    ports:
      - protocol: TCP
        port: 24224
