   name            =  "allow_ssh_http" 
   description     = "Allow SSH and HTTP access"
   ingress {
     front_port  = 22
     to_port     = 22
     protocol    = "tcp"
     cidr_block  = ["0.0.0.0/0"]
}

   ingress {
     front_port     = 80
     to_port        = 80

Jenkinsfile - docker build and deploy

pipeline {
   agent any 
     stages {
        stage('Build docker image') {
            steps {
               script {
                  docker.build('Ankit-app')
                }
             }
         }
       stage('Push docker image') {
           steps {
             script {
                docker with registry('https://docker.io', 'docker-credentials') {
                    docker.image('akit-app').push('latest')
                           }
                        }
                    }
                }
          }
     }  


GitLab CI/CD - Build & Push Docker Image
 stages:
    - build
    - push

  build:
     stage: build
     script:
         - docker build -t ankit-app
  push:
    stage: push
     script
        - docker login -u $CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
        - docker push ankit-app
