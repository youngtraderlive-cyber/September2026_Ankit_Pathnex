Kubernetes disaster recovery with persistent volume
apiVersion: app/v1
kind: Statefulset
metadata:
  name: Ankit-statefulset
spec:
  serviceName: "Ankit"
  replicas: 3
  selector: 
   matchLabels:
    app: Ankit
  templare:
   metadata:
     labels:
       app: Ankit
   spec: 
     container:
        - name: nginx
          image: nginx
          volumeMounts:
             - mountPath: /user/share/nginx/html
               name: web-content

   volumeClaimTemplate:
     - metadata:
         name: web-content
     spec:
        accessModes: [ "ReadWriteOnce" ]
        resource:
         request:
           storage: 5Gi
