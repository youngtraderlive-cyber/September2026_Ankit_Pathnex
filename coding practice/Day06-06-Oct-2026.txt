Ansible Install Docker

- name: Install docker
  hosts: all
  become: yes
    tasks:
      - name: Install docker
          yum:
            name: docker
            state:present
      - name: Start docker
        service: 
            name: docker
            state: started
             enabled: yes

kubernetes pod with volume mount
apiVersion: v1
kind: pod
metadata:
  name: ankit-pod
spec: 
   container:
     - name: web
       image: Nginx
       volumemount
          - mountPath: /var/share/nginx/html
            name: html-volume
    volume:
      - nameL html-volume
        hostPath:
           path: /data
           type: Directory
