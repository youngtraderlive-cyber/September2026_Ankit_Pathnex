Security hardening and compliance
- name: configure firewall with iptables
  hosts: all
  become: yes
     tasks:
       - name set default policies to drop all traffic
         command: iptables -p input drop
       - name: allow ssh traffic
         commandL: iptables -A INPUT -p tcp --drop 22 -j ACCEPT
       - name: allow HTTP traffic
         command: iptables -A Input -p tcp --dropt 80 -j ACCEPT
       - name: Save iptables configuration
         command: service iptables save

    - build
    - push
    - deploy

    stage: build
    script:
        - docker build -t ankit-web-app

   stage: push
   script:
       - docker login -u "CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
       - docker push ankit-web-app
deploy:
    stage: deploy
    script:
           - kubectl apply - kubernetes/deployment.yaml
