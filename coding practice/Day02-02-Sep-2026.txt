Ansible - Install & Enable Nginx
- name: Install and status Nginx on Pathnex
  hosts: all
  become: yes

  tasks:
   - name: Install nginx
     yum:
      name: nginx
      state: present

    - name: Enable nginx
      service:
         name: nginx
         state: started
         enabled: yes
