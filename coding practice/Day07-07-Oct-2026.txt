- name: setup nginx with custom config
  hosts: all 
  become: yes
    taks:
      - name: Install nginx
         yum:
          name: Nginx
          state: enable
      - name: Copy custom nginx.conf
        copy:
          src: ngix.config
          dest: /ect/nginx/nginx.conf
      - name: Start nginx service
        service: 
         name: nginx
         state: started
         enabled: yes
