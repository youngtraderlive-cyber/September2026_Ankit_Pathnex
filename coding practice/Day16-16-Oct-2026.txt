Ansible - Craate and Start EC2 Insatce

- name: Create EC2 Instance
  hosts: localHost
  become: yes
    tasks:
       - name: Launch EC2 instance in us-east-1
          ec2_instace:
            name: ankit-ec2-us-east-1
            key_name: ankit-key
            region: us-east-1
            instance_type: t2.micro
            wait: yes
       - name: Launch EC2 instance in us-west-1
          ec2_instance:
             name: ankit-ec2-us-west-1
             key_name: ankit-key
             instance_type: t2.micro
             image: ami-0hgfdhgyuftfvv1234
             wait: yes
