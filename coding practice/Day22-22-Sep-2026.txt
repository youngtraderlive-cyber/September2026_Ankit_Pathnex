- name: deploy nginx acress multiple kubernetes clusters
  hosts: localhost
  tasks:
     - name: configure kubectl for cluster 1
       command: kubectl --conext=us-est confir use-context us-east-cluster
     - name: deploy nginx on cluster 1
       command: kubectl apply -f kybernetes/deployment.yaml
     - name: configure kubectl for cluster 2
       command: kubectl --context=us-west config use-context us-west-cluster
     - name: deploy nginx on cluster 2
       command: kubectl apply -f kubenetes/deployment.yaml
