Ansible 
- name: Setup Nginx with Custom Config
  hosts: all
  become: yes
  task:
    - name: Install nginx
      yum:
       name: nginx
       state: present
 - name: Copy custom nginx.conf
   copy: 
    src: nginx.conf
    dest: /etc/nginx/nginx.conf
- name: Start nginx serice
  service:
    name: nginx
    state: started
    enabled: yes
