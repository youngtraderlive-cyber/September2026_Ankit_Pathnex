kubernetes helm-based Auto-Scalling Deployment

apiVersion: app/v1
kind: deployment
 metadata:
    - name: ankit-web-app
spec:
   replicas: 3
   
selector:
   matchLabels:
      app: ankit-web-app
temlate:
  metadata:
     labels:
       app: ankit-web-app

spec: 
  container:
     - name: nginx
       image: nginx
       ports:
         - containerPort: 80

---


apiVersion: autoscaling/v2
kind: Horizontalpodautoscaler
metadata:
    - name: ankit-web-app-hpa
spec:
   scaleTargetRef:
      apiVersion: apps/v1
      kind: deployment
      name: ankit-web-app
  minReplicas: 2
  maxReplicas: 10
  targetCPUUtilizationPercentage: 80
