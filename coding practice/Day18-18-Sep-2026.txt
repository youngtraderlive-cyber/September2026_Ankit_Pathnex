Ansible configure HAproxy for load balancing
- name: Setup HAproxy Load Balancer
  hosts: all
  become: yes
    tasks:
      - name: Install Haproxy
        yum:
         name: haproxy
         state: present
      - name: Configure HAProxy
        template:
          src: haproxy.cfg.k2
          dest: /etc/haproxy/haproxy.cfg
     - name: Start HAproxy Service
       service:
         name: haproxy
         state: started
         enabled: yes
